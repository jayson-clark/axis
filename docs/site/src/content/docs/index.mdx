import { Card, CardGrid } from '@astrojs/starlight/components';

```axis
config { showGrid: true; degreeMode: false }

style bold { lineWidth: 4 }
macro bell(s) = e ^ (-x ^ 2 / s)

"Basic functions"

f(x) = x ^ 2 - 4x + 3 @ color: RED
g(x) = sin(x) + cos(2x) @ color: BLUE, use: bold

a = 1 @ slider: 0.5..4 step 0.5
h(x) = bell(a) @{
    color: GREEN
    lineStyle: DASHED
    label: "a bell"
}

n = 0
ticker n -> n + dt / 1000 @ playing
```

    <Card title="Text, not clicks" icon="document">

    <Card title="Styling that stays put" icon="pencil">

    </Card>

        calculator before it is published.
